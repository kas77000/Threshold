# CEPT (Conditional Expected-Performance Threshold) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the `cept` package (`check` / `fit` / `score`). It flags HK TWAP/VWAP orders whose `ePvwap/Sprd` is abnormal given the order's own conditions, at policy-fixed rarities, and produces the evidence pack (backtest, detection power, volatility sensitivity, cohort test) described in `docs/superpowers/specs/2026-11-15-conditional-threshold-design.md`.

**Architecture:**
1. A loader with row accounting, then record-level data-quality rules.
2. Per benchmark cell, a frozen location–scale model: Huber centre, Huber regression of `log|residual|` for dispersion.
3. A robustly standardised conditional z.
4. Per-side thresholds `MAX(floor, quantile-matched Student-t quantile at the policy rarity)` for Tier A (1/10k, floor 4) and Tier B (1/1k, floor 3).
5. Monitoring modules (LOMO backtest with Kupiec p-values, injected detection, vol sensitivity, cohort test, drift) built on the same scoring function.

**Tech Stack:** Python 3.13, numpy 2.x, pandas 2.x, scipy 1.15, scikit-learn 1.6 (HuberRegressor), matplotlib, tabulate (for `to_markdown`), pytest 8.

- `Vol` = single-day Parkinson volatility (daily fraction). `Dur` in minutes. `PR`, `%Adv` and `FR` in percent. `#Shares` = ordered quantity.
- Data-quality rules test the record, never the performance value.
- Row accounting invariant: `rows_in == rows_kept + sum(dropped)`.
- Scoring a window that overlaps the fit window is refused (`GuardError`).
- JSON artifacts are written with `allow_nan=False`. NaN is stored as null.
- Run tests from the repo root: `python -m pytest -q`. `pytest.ini` sets `pythonpath = .`.

```python
"""Read the extract and account for every row that leaves."""

from __future__ import annotations

import os
from dataclasses import dataclass, field

import numpy as np
import pandas as pd

from cept import config, schema

MISSING_ORDER_ID = "missing order id"
MISSING_SYMBOL = "missing symbol"
MISSING_DATE = "missing or unparseable date"
UNMAPPED_STRATEGY = "unmapped strategy"
OTHER_MARKET = "other market"
MISSING_METRIC = "missing metric"
_REASONS = [MISSING_ORDER_ID, MISSING_SYMBOL, MISSING_DATE,
            UNMAPPED_STRATEGY, OTHER_MARKET, MISSING_METRIC]


@dataclass
class CleaningReport:
    rows_in: int = 0
    rows_kept: int = 0
    dropped: dict = field(default_factory=dict)
    unmapped_strategies: dict = field(default_factory=dict)
    date_min: pd.Timestamp | None = None
    date_max: pd.Timestamp | None = None

    def to_frame(self) -> pd.DataFrame:
        rows = [{"reason": r, "rows": n} for r, n in self.dropped.items()]
        return pd.DataFrame(rows, columns=["reason", "rows"])

    def summary(self) -> str:
        lines = [f"rows in  : {self.rows_in:,}", f"rows kept: {self.rows_kept:,}"]
        lines += [f"  dropped ({r}): {n:,}" for r, n in self.dropped.items() if n]
        if self.unmapped_strategies:
            lines.append("  UNMAPPED STRATEGIES (excluded, not defaulted): "
                         + ", ".join(f"{s} ({n})" for s, n in
                                     sorted(self.unmapped_strategies.items())))
        if self.date_min is not None:
            lines.append(f"date range: {self.date_min.date()} .. "
                         f"{self.date_max.date()}")
        return "\n".join(lines)


def read_any(path: str) -> pd.DataFrame:
    ext = os.path.splitext(path)[1].lower()
    if ext == ".csv":
        return pd.read_csv(path)
    if ext in (".xlsx", ".xls"):
        return pd.read_excel(path)
    if ext == ".parquet":
        return pd.read_parquet(path)
    raise ValueError(f"Unsupported extension '{ext}'. Use .csv, .xlsx or .parquet.")


def prepare(raw: pd.DataFrame, market: str | None = None
            ) -> tuple[pd.DataFrame, CleaningReport]:
    inverse = {v: k for k, v in config.COLUMN_MAP.items()}
    missing = [inverse[c] for c in schema.REQUIRED if inverse[c] not in raw.columns]
    if missing:
        raise ValueError("Extract is missing required column(s): "
                         + ", ".join(missing))

    rep = CleaningReport(rows_in=len(raw), dropped={r: 0 for r in _REASONS})
    present = {k: v for k, v in config.COLUMN_MAP.items() if k in raw.columns}
    df = raw.rename(columns=present)[list(present.values())].copy()

    sym = df[schema.SYMBOL].astype(str).str.strip()
    df[schema.MARKET] = sym.str[-2:].str.upper()
    df.loc[sym.isin(["", "nan", "None"]), schema.MARKET] = pd.NA
    df[schema.BENCHMARK] = df[schema.ALGO].map(config.benchmark_for)
    df[schema.ORDER_DATE] = pd.to_datetime(df[schema.ORDER_DATE],
                                           errors="coerce").dt.normalize()
    for col in schema.NUMERIC:
        if col in df.columns:
            df[col] = pd.to_numeric(df[col], errors="coerce")
    df[schema.METRIC] = df[schema.METRIC].replace([np.inf, -np.inf], np.nan)

    oid = df[schema.ORDER_ID].astype(str).str.strip()
    other_mkt = (pd.Series(False, index=df.index) if market is None
                 else df[schema.MARKET] != str(market).upper())
    checks = [
        (MISSING_ORDER_ID, df[schema.ORDER_ID].isna() | oid.isin(["", "nan", "None"])),
        (MISSING_SYMBOL, df[schema.MARKET].isna()),
        (MISSING_DATE, df[schema.ORDER_DATE].isna()),
        (UNMAPPED_STRATEGY, df[schema.BENCHMARK].isna()),
        (OTHER_MARKET, other_mkt),
        (MISSING_METRIC, df[schema.METRIC].isna()),
    ]
    alive = pd.Series(True, index=df.index)
    for reason, failed in checks:
        hit = alive & failed.fillna(True).astype(bool)
        rep.dropped[reason] = int(hit.sum())
        if reason == UNMAPPED_STRATEGY and hit.any():
            vc = df.loc[hit, schema.ALGO].astype(str).str.strip().value_counts()
            rep.unmapped_strategies = {str(k): int(v) for k, v in vc.items()}
        alive &= ~hit

    df = df[alive].reset_index(drop=True)
    rep.rows_kept = len(df)
    if len(df):
        rep.date_min = df[schema.ORDER_DATE].min()
        rep.date_max = df[schema.ORDER_DATE].max()
    return df, rep


def load(path: str, market: str | None = None):
    return prepare(read_any(path), market=market)
```

```python
"""Record-level data quality. Rules test the RECORD, never the performance.

A rule of the form "performance too large" would be a hidden second
threshold. Every rule here is about whether the record can be true at all.
Rules are checked in order; a row is reported under its first failure.
"""

from __future__ import annotations

import numpy as np
import pandas as pd

from cept import config, schema as s

_NUM_CONDITIONS = [c for c in s.CONDITIONS if c != s.SIDE]


def _col(df, c):
    return df[c] if c in df.columns else pd.Series(np.nan, index=df.index)


RULES = {
    "condition_missing": lambda d: d[_NUM_CONDITIONS].isna().any(axis=1)
                                   | d[s.SIDE].isna(),
    "spread_implausible": lambda d: ~d[s.SPREAD_BPS].between(*config.SPREAD_RANGE_BPS),
    "metric_inconsistent": lambda d: (
        _col(d, s.SLIPPAGE_BPS).notna()
        & ((_col(d, s.SLIPPAGE_BPS) / d[s.SPREAD_BPS] - d[s.METRIC]).abs()
           > config.METRIC_CONSISTENCY_TOL * np.maximum(1.0, d[s.METRIC].abs()))),
    "participation_impossible": lambda d: ~((d[s.PARTICIPATION] > 0)
                                            & (d[s.PARTICIPATION] <= 100)),
    "duration_nonpositive": lambda d: ~(d[s.DURATION_MIN] > 0),
    "auction_share_impossible": lambda d: (d[s.OPEN_PCT] + d[s.CLOSE_PCT]) > 100.0,
    "adv_nonpositive": lambda d: ~(d[s.PCT_ADV] > 0),
    "volatility_nonpositive": lambda d: ~(d[s.VOLATILITY] > 0),
    "fill_rate_impossible": lambda d: _col(d, s.FILL_RATE).notna()
                                      & ~((_col(d, s.FILL_RATE) > 0)
                                          & (_col(d, s.FILL_RATE) <= 100)),
}


def reasons(df: pd.DataFrame) -> pd.Series:
    out = pd.Series("", index=df.index, dtype=object)
    for name, rule in RULES.items():
        hit = rule(df).fillna(True).astype(bool) & (out == "")
        out[hit] = name
    return out


def split(df: pd.DataFrame) -> tuple[pd.DataFrame, pd.DataFrame]:
    r = reasons(df)
    exc = df[r != ""].copy()
    exc["dq_reason"] = r[r != ""]
    return df[r == ""].reset_index(drop=True), exc.reset_index(drop=True)
```

Note on rule order: `condition_missing` runs first, so a NaN never reaches a later comparison. The test row 8 (`Dur = NaN`) reports `condition_missing`, while row 3 (`Dur = 0`) reports `duration_nonpositive`. `fillna(True)` is a backstop.

```python
"""Order conditions -> model inputs.

Volatility is split into the stock's TYPICAL level (median of its own
same-day Parkinson vol over the fit year) and the day's SURPRISE relative to
it. Same-day vol is partly caused by a large order itself; keeping the
surprise as its own term is what lets sensitivity.py show exactly which
alerts it clears.
"""

from __future__ import annotations

import numpy as np
import pandas as pd

from cept import schema as s

FEATURE_NAMES = ["log_vol_typ", "log_vol_surprise", "log_dur", "log_spread",
                 "log_pct_adv", "sqrt_pr", "open_frac", "close_frac", "is_buy"]


def feature_names(include_surprise: bool) -> list[str]:
    return [n for n in FEATURE_NAMES
            if include_surprise or n != "log_vol_surprise"]


class VolTypical:
    def __init__(self, by_symbol: dict, fallback: float):
        self.by_symbol = dict(by_symbol)
        self.fallback = float(fallback)

    @classmethod
    def fit(cls, df: pd.DataFrame, min_n: int) -> "VolTypical":
        g = df.groupby(s.SYMBOL)[s.VOLATILITY]
        n, med = g.size(), g.median()
        keep = med[n >= min_n]
        return cls({str(k): float(v) for k, v in keep.items()},
                   float(df[s.VOLATILITY].median()))

    def lookup(self, df: pd.DataFrame) -> np.ndarray:
        return (df[s.SYMBOL].astype(str).map(self.by_symbol)
                .fillna(self.fallback).to_numpy(dtype=float))

    def to_dict(self) -> dict:
        return {"by_symbol": self.by_symbol, "fallback": self.fallback}

    @classmethod
    def from_dict(cls, d: dict) -> "VolTypical":
        return cls(d["by_symbol"], d["fallback"])


def design(df: pd.DataFrame, voltyp: VolTypical,
           include_surprise: bool = True) -> np.ndarray:
    vt = voltyp.lookup(df)
    vol = df[s.VOLATILITY].to_numpy(dtype=float)
    cols = [np.log(vt)]
    if include_surprise:
        cols.append(np.log(vol / vt))
    cols += [
        np.log(df[s.DURATION_MIN].to_numpy(dtype=float)),
        np.log(df[s.SPREAD_BPS].to_numpy(dtype=float)),
        np.log(df[s.PCT_ADV].to_numpy(dtype=float)),
        np.sqrt(df[s.PARTICIPATION].to_numpy(dtype=float) / 10.0),
        df[s.OPEN_PCT].to_numpy(dtype=float) / 100.0,
        df[s.CLOSE_PCT].to_numpy(dtype=float) / 100.0,
        df[s.SIDE].astype(str).str.upper().str.startswith("B")
          .to_numpy(dtype=float),
    ]
    return np.column_stack(cols)
```

```python
"""Expected performance and expected dispersion given order conditions.

centre : Huber M-regression (Huber 1964). Bounded influence, so a bad order
         cannot drag the expectation toward itself.
scale  : Huber regression of log|residual| on the same conditions -- the
         multiplicative heteroskedasticity model (Harvey 1976). One constant
         then rescales so that median |z| matches the normal (0.6745).
Two passes; the second centre fit is weighted by 1/scale^2.

Coefficients are stored as plain arrays so the frozen model is a JSON file
that any reader can apply with a dot product.
"""

from __future__ import annotations

import numpy as np
from sklearn.linear_model import HuberRegressor


def _huber():
    return HuberRegressor(epsilon=1.35, alpha=0.0, max_iter=1000)


class LocationScale:
    def fit(self, X: np.ndarray, y: np.ndarray, n_pass: int = 2):
        X = np.asarray(X, float)
        y = np.asarray(y, float)
        self.x_mean = X.mean(axis=0)
        self.x_sd = X.std(axis=0)
        self.x_sd[self.x_sd == 0] = 1.0
        Xs = (X - self.x_mean) / self.x_sd
        w = np.ones(len(y))
        for _ in range(n_pass):
            loc = _huber().fit(Xs, y, sample_weight=w)
            r = y - loc.predict(Xs)
            sc = _huber().fit(Xs, np.log(np.abs(r) + 1e-9))
            s = np.exp(sc.predict(Xs))
            w = 1.0 / s ** 2
            w /= w.mean()
        self.loc_coef = loc.coef_.copy()
        self.loc_intercept = float(loc.intercept_)
        self.sc_coef = sc.coef_.copy()
        self.sc_intercept = float(sc.intercept_)
        z = r / s
        self.c = float(np.median(np.abs(z - np.median(z))) / 0.6745)
        return self

    def predict(self, X):
        Xs = (np.asarray(X, float) - self.x_mean) / self.x_sd
        mu = Xs @ self.loc_coef + self.loc_intercept
        s = np.exp(Xs @ self.sc_coef + self.sc_intercept) * self.c
        return mu, s

    def z(self, X, y):
        mu, s = self.predict(X)
        return (np.asarray(y, float) - mu) / s

    def to_dict(self) -> dict:
        return {k: (v.tolist() if isinstance(v, np.ndarray) else v)
                for k, v in {"x_mean": self.x_mean, "x_sd": self.x_sd,
                             "loc_coef": self.loc_coef,
                             "loc_intercept": self.loc_intercept,
                             "sc_coef": self.sc_coef,
                             "sc_intercept": self.sc_intercept,
                             "c": self.c}.items()}

    @classmethod
    def from_dict(cls, d: dict) -> "LocationScale":
        m = cls()
        for k in ("x_mean", "x_sd", "loc_coef", "sc_coef"):
            setattr(m, k, np.asarray(d[k], float))
        m.loc_intercept, m.sc_intercept, m.c = (float(d["loc_intercept"]),
                                                float(d["sc_intercept"]),
                                                float(d["c"]))
        return m
```

```python
"""One side's tail, from a Student-t fitted by QUANTILE MATCHING.

Only the body and the moderate tail (up to the 99.5th percentile of the
side) inform the fit, so the year's own extreme orders -- the ones the band
exists to catch -- cannot widen the threshold that judges them. In the lab,
maximum-likelihood t and GPD peaks-over-threshold fits were both inflated by
0.1% contamination; this one was not.

Convention: t is the side's signed distance (z for the high side, -z for the
low side), centred on the median, so P(t > 0) ~ 1/2. P(T > x | T > 0) is
2 * sf(x), hence matching at probability 1 - (1 - p)/2.
"""

from __future__ import annotations

from dataclasses import asdict, dataclass

import numpy as np
from scipy import stats
from scipy.optimize import minimize

from cept import config


@dataclass(frozen=True)
class TailFit:
    df: float
    scale: float

    def to_dict(self) -> dict:
        return asdict(self)

    @classmethod
    def from_dict(cls, d: dict) -> "TailFit":
        return cls(float(d["df"]), float(d["scale"]))


def fit_side(t, probs=config.TAIL_PROBS) -> TailFit:
    t = np.asarray(t, float)
    half = t[t > 0]
    q_emp = np.quantile(half, probs)
    pp = 1.0 - (1.0 - np.asarray(probs)) / 2.0

    def loss(theta):
        nu, sc = np.exp(theta)
        q = stats.t.ppf(pp, nu + 2.0, 0.0, sc)
        return float(np.sum((np.log(q) - np.log(q_emp)) ** 2))

    best = minimize(loss, x0=np.log([8.0, 1.0]), method="Nelder-Mead",
                    options={"xatol": 1e-6, "fatol": 1e-10, "maxiter": 4000})
    nu, sc = np.exp(best.x)
    return TailFit(df=float(nu + 2.0), scale=float(sc))


def threshold(tf: TailFit, rarity: float, floor: float) -> float:
    return float(max(floor, stats.t.isf(rarity / 2.0, tf.df, 0.0, tf.scale)))


def exceed_prob(tf: TailFit, thr: float) -> float:
    return float(stats.t.sf(thr, tf.df, 0.0, tf.scale))
```

```python
    @classmethod
    def from_dict(cls, d: dict) -> "CellModel":
        return cls(d["benchmark"], int(d["n"]), bool(d["include_surprise"]),
                   VolTypical.from_dict(d["voltyp"]),
                   LocationScale.from_dict(d["ls"]),
                   float(d["z_med"]), float(d["z_mad"]),
                   TailFit.from_dict(d["tail_hi"]),
                   TailFit.from_dict(d["tail_lo"]), d["thr"],
                   d.get("cohort_edges", {}), d.get("medians", {}))


def fit_cell(df: pd.DataFrame, benchmark: str,
             include_surprise: bool = True) -> CellModel:
    vt = VolTypical.fit(df, config.MIN_SYMBOL_N)
    X = design(df, vt, include_surprise)
    y = df[s.METRIC].to_numpy(dtype=float)
    ls = LocationScale().fit(X, y)
    z0 = ls.z(X, y)
    med = float(np.median(z0))
    mad = float(1.4826 * np.median(np.abs(z0 - med)))
    z = (z0 - med) / mad
    t_hi, t_lo = fit_side(z), fit_side(-z)
    thr = {tier: {"hi": threshold(t_hi, r, f), "lo": threshold(t_lo, r, f)}
           for tier, (r, f) in TIERS.items()}
    qs = np.linspace(0, 1, config.COHORT_BINS + 1)[1:-1]
    edges = {f: np.unique(np.quantile(df[f].dropna(), qs)).tolist()
             for f in config.COHORT_FEATURES
             if f in df.columns and df[f].notna().any()}
    medians = {f: float(df[f].median()) for f in s.DRIFT_FEATURES
               if f in df.columns and df[f].notna().any()}
    return CellModel(benchmark, len(df), include_surprise, vt, ls, med, mad,
                     t_hi, t_lo, thr, edges, medians)


def fit_all(df: pd.DataFrame, include_surprise: bool = True,
            min_cell_n: int = config.MIN_CELL_N):
    cells, skipped = {}, {}
    for bench, g in df.groupby(s.BENCHMARK):
        if len(g) < min_cell_n:
            skipped[str(bench)] = int(len(g))
            continue
        cells[str(bench)] = fit_cell(g, str(bench), include_surprise)
    return cells, skipped


def zscores(cm: CellModel, df: pd.DataFrame, y=None):
    X = design(df, cm.voltyp, cm.include_surprise)
    mu, sc = cm.ls.predict(X)
    y = df[s.METRIC].to_numpy(dtype=float) if y is None else np.asarray(y, float)
    z = ((y - mu) / sc - cm.z_med) / cm.z_mad
    return z, mu + cm.z_med * sc, sc * cm.z_mad


def predicted_count(cm: CellModel, n: int, tier: str) -> float:
    t = cm.thr[tier]
    return float(n * (exceed_prob(cm.tail_hi, t["hi"])
                      + exceed_prob(cm.tail_lo, t["lo"])))
```

```python
"""Apply frozen cell models to a month. Refuse what would be meaningless."""

from __future__ import annotations

import numpy as np
import pandas as pd

from cept import config, schema as s
from cept.fit import zscores


class GuardError(ValueError):
    """A refusal. Never downgraded to a warning."""


def check_no_overlap(fit_start, fit_end, df: pd.DataFrame) -> None:
    d = df[s.ORDER_DATE].dropna()
    if d.empty or fit_start is None or fit_end is None:
        return
    lo, hi = d.min(), d.max()
    if lo <= pd.Timestamp(fit_end) and pd.Timestamp(fit_start) <= hi:
        raise GuardError(
            f"Scored window {lo.date()}..{hi.date()} overlaps the fit window "
            f"{fit_start}..{fit_end}; the flag rate would be circular.")


def _tags(df: pd.DataFrame, voltyp_lookup: np.ndarray) -> list[str]:
    out = []
    fr = df[s.FILL_RATE] if s.FILL_RATE in df else pd.Series(np.nan, index=df.index)
    surprise = df[s.VOLATILITY].to_numpy(float) / voltyp_lookup
    for i, (_, r) in enumerate(df.iterrows()):
        t = []
        if pd.notna(fr.iloc[i]) and fr.iloc[i] < config.PARTIAL_FILL_BELOW:
            t.append(f"partial fill (FR {fr.iloc[i]:.0f}%)")
        if surprise[i] > config.VOL_SURPRISE_ABOVE:
            t.append(f"volatile day ({surprise[i]:.1f}x typical)")
        if r[s.OPEN_PCT] + r[s.CLOSE_PCT] > config.AUCTION_HEAVY_ABOVE:
            t.append("auction heavy")
        if r[s.PARTICIPATION] > config.HIGH_PR_ABOVE:
            t.append(f"high participation ({r[s.PARTICIPATION]:.0f}%)")
        if r[s.PCT_ADV] > config.HIGH_ADV_PCT:
            t.append(f"large vs ADV ({r[s.PCT_ADV]:.0f}%)")
        out.append("; ".join(t))
    return out


def score_orders(cells: dict, df: pd.DataFrame) -> pd.DataFrame:
    out = df.copy()
    for c in ("expected_perf", "expected_disp", "z", "excess"):
        out[c] = np.nan
    out["tier"] = "NO_MODEL"
    out["tags"] = ""
    for bench, g in df.groupby(s.BENCHMARK):
        cm = cells.get(str(bench))
        if cm is None:
            continue
        z, mu, disp = zscores(cm, g)
        a, b = cm.thr["A"], cm.thr["B"]
        tier = np.where((z > a["hi"]) | (z < -a["lo"]), "A",
                        np.where((z > b["hi"]) | (z < -b["lo"]), "B", "IN_RANGE"))
        excess = np.where(tier == "A",
                          np.where(z > 0, z - a["hi"], -z - a["lo"]), np.nan)
        out.loc[g.index, "expected_perf"] = mu
        out.loc[g.index, "expected_disp"] = disp
        out.loc[g.index, "z"] = z
        out.loc[g.index, "tier"] = tier
        out.loc[g.index, "excess"] = excess
        out.loc[g.index, "tags"] = _tags(g, cm.voltyp.lookup(g))
    return out


def drift(cells: dict, df: pd.DataFrame) -> pd.DataFrame:
    rows = []
    for bench, cm in cells.items():
        g = df[df[s.BENCHMARK] == bench]
        for f, fit_med in cm.medians.items():
            mm = float(g[f].median()) if f in g and g[f].notna().any() else np.nan
            pct = (100.0 * (mm - fit_med) / abs(fit_med)
                   if np.isfinite(mm) and fit_med else np.nan)
            rows.append({"benchmark": bench, "feature": f, "fit_median": fit_med,
                         "month_median": mm, "pct_change": pct,
                         "warn": bool(np.isfinite(pct)
                                      and abs(pct) > config.DRIFT_THRESHOLD_PCT)})
    return pd.DataFrame(rows, columns=["benchmark", "feature", "fit_median",
                                       "month_median", "pct_change", "warn"])
```

```python
"""Systematic deterioration that no single order reveals.

Mean conditional z (winsorised at +/-4, so one outlier cannot drive a cohort)
per strategy and per quintile of each condition, quintile edges from the FIT
year, tested against zero. The model is frozen at fit, so execution that gets
worse afterwards -- even when correlated with a covariate -- shows up here.
"""

from __future__ import annotations

import numpy as np
import pandas as pd

from cept import config, schema as s

COLS = ["benchmark", "feature", "bucket", "n", "mean_z", "t", "alert"]


def cohort_test(cells: dict, scored: pd.DataFrame,
                t_crit: float = config.COHORT_T) -> pd.DataFrame:
    rows = []
    for bench, g in scored[scored["tier"] != "NO_MODEL"].groupby(s.BENCHMARK):
        cm = cells[str(bench)]
        z = np.clip(g["z"].to_numpy(float), -4.0, 4.0)
        groups = [("ALL", "all", np.ones(len(g), bool))]
        for f, edges in cm.cohort_edges.items():
            if f not in g:
                continue
            bins = np.digitize(g[f].to_numpy(float), edges)
            groups += [(f, int(i), bins == i) for i in np.unique(bins)]
        for f, b, mask in groups:
            n = int(mask.sum())
            if n < config.COHORT_MIN_N:
                continue
            zz = z[mask]
            t = float(zz.mean() / (zz.std(ddof=1) / np.sqrt(n)))
            rows.append({"benchmark": str(bench), "feature": f, "bucket": b,
                         "n": n, "mean_z": float(zz.mean()), "t": t,
                         "alert": abs(t) > t_crit})
    return pd.DataFrame(rows, columns=COLS)
```

```python
"""Is the stated rarity true? Predicted vs realised exceedances.

LOMO: each month of the fit year is scored by models fitted on the other
months -- out-of-sample, as in production. Kupiec's proportion-of-failures
likelihood ratio gives a p-value for realised vs stated rate.
"""

from __future__ import annotations

import numpy as np
import pandas as pd
from scipy import stats

from cept import config, schema as s
from cept.fit import fit_all, predicted_count
from cept.score import score_orders

COLS = ["benchmark", "n", "pred_A", "real_A", "p_A", "pred_B", "real_B", "p_B"]


def kupiec_pvalue(n: int, x: int, p: float) -> float:
    if n <= 0:
        return float("nan")
    phat = x / n

    def ll(q):
        a = (n - x) * np.log1p(-q) if x < n else 0.0
        b = x * np.log(q) if x > 0 else 0.0
        return a + b
    lr = -2.0 * (ll(p) - (ll(phat) if 0 < phat < 1 else 0.0))
    return float(stats.chi2.sf(max(lr, 0.0), 1))


def month_table(cells: dict, scored: pd.DataFrame) -> pd.DataFrame:
    rows = []
    for bench, g in scored[scored["tier"] != "NO_MODEL"].groupby(s.BENCHMARK):
        cm = cells[str(bench)]
        n = len(g)
        pa, pb = predicted_count(cm, n, "A"), predicted_count(cm, n, "B")
        ra = int((g["tier"] == "A").sum())
        rb = int(g["tier"].isin(["A", "B"]).sum())
        rows.append({"benchmark": str(bench), "n": n,
                     "pred_A": pa, "real_A": ra,
                     "p_A": kupiec_pvalue(n, ra, pa / n),
                     "pred_B": pb, "real_B": rb,
                     "p_B": kupiec_pvalue(n, rb, pb / n)})
    return pd.DataFrame(rows, columns=COLS)


def lomo(df: pd.DataFrame, include_surprise: bool = True,
         min_cell_n: int = config.MIN_CELL_N) -> pd.DataFrame:
    month = df[s.ORDER_DATE].dt.to_period("M").astype(str)
    out = []
    for label in sorted(month.unique()):
        held = month == label
        cells, _ = fit_all(df[~held], include_surprise, min_cell_n)
        t = month_table(cells, score_orders(cells, df[held]))
        t.insert(0, "month", label)
        out.append(t)
    return pd.concat(out, ignore_index=True) if out else pd.DataFrame(
        columns=["month"] + COLS)
```

```python
"""Injected detection: what size of miss does the frozen model catch?

Take real orders, worsen (or improve) each by exactly m of its OWN expected
dispersion, and score. Detection then depends on each order's actual noise,
not on an assumed distribution -- this is the synthetic-data calibration test
FCA Market Watch 86 describes, run on the firm's own book.
"""

from __future__ import annotations

import numpy as np
import pandas as pd

from cept import config, schema as s
from cept.fit import zscores

COLS = ["benchmark", "size", "side", "detected_A", "detected_B"]


def injected_detection(cells: dict, df: pd.DataFrame,
                       sizes=config.INJECT_SIZES, n: int = config.INJECT_N,
                       seed: int = 0) -> pd.DataFrame:
    rng = np.random.default_rng(seed)
    rows = []
    for bench, cm in cells.items():
        g = df[df[s.BENCHMARK] == bench]
        g = g.iloc[rng.choice(len(g), min(n, len(g)), replace=False)]
        _, _, disp = zscores(cm, g)
        y = g[s.METRIC].to_numpy(float)
        for m in sizes:
            for side, sign in (("lo", -1.0), ("hi", 1.0)):
                z, _, _ = zscores(cm, g, y + sign * m * disp)
                if side == "lo":
                    da, db = z < -cm.thr["A"]["lo"], z < -cm.thr["B"]["lo"]
                else:
                    da, db = z > cm.thr["A"]["hi"], z > cm.thr["B"]["hi"]
                rows.append({"benchmark": bench, "size": int(m), "side": side,
                             "detected_A": float(da.mean()),
                             "detected_B": float(db.mean())})
    return pd.DataFrame(rows, columns=COLS)
```

```python
"""Which alerts does the same-day volatility term clear?

Same-day Parkinson vol is partly caused by the order itself. This refits the
model WITHOUT the day-surprise term and lists every order that would then be
Tier A but is not under the production model -- with the large-vs-ADV ones
marked, so it is visible whether any order excused itself.
"""

from __future__ import annotations

import pandas as pd

from cept import config, schema as s
from cept.fit import fit_all
from cept.score import score_orders

COLS = ["order_id", "benchmark", "perf_in_spreads", "z_with", "z_without",
        "vol_surprise", "pct_adv", "high_adv"]


def vol_sensitivity(fit_df: pd.DataFrame, target_df: pd.DataFrame,
                    cells: dict) -> pd.DataFrame:
    without, _ = fit_all(fit_df, include_surprise=False, min_cell_n=1)
    a = score_orders(cells, target_df)
    b = score_orders({k: without[k] for k in cells if k in without}, target_df)
    hit = (b["tier"] == "A") & (a["tier"] != "A")
    rows = []
    for i in target_df.index[hit.to_numpy()]:
        cm = cells[target_df.loc[i, s.BENCHMARK]]
        vt = cm.voltyp.lookup(target_df.loc[[i]])[0]

                     "benchmark": target_df.loc[i, s.BENCHMARK],
                     "perf_in_spreads": float(target_df.loc[i, s.METRIC]),
                     "z_with": float(a.loc[i, "z"]),
                     "z_without": float(b.loc[i, "z"]),
                     "vol_surprise": float(target_df.loc[i, s.VOLATILITY] / vt),
                     "pct_adv": adv, "high_adv": adv > config.HIGH_ADV_PCT})

def summary(df: pd.DataFrame) -> dict:
```

```python
"""The frozen model file: everything needed to score -- and to explain why."""

from __future__ import annotations

import hashlib
import json
import math
from datetime import datetime, timezone

from cept import config
from cept.fit import CellModel

MODEL_VERSION = 1
_POLICY_KEYS = ["RARITY_A", "FLOOR_A", "RARITY_B", "FLOOR_B", "COHORT_T",
                "POLICY_RATIONALE", "MIN_CELL_N", "MIN_SYMBOL_N", "TAIL_PROBS"]


def policy_snapshot() -> dict:
    return {k: getattr(config, k) for k in _POLICY_KEYS}


def file_hash(path) -> str:
    h = hashlib.sha256()
    with open(path, "rb") as fh:
        for chunk in iter(lambda: fh.read(1 << 20), b""):
            h.update(chunk)
    return h.hexdigest()[:16]


def _clean(o):
    if isinstance(o, dict):
        return {str(k): _clean(v) for k, v in o.items()}
    if isinstance(o, (list, tuple)):
        return [_clean(v) for v in o]
    if isinstance(o, float) and not math.isfinite(o):
        return None
    if hasattr(o, "item"):            # numpy scalar
        return _clean(o.item())
    return o


def write_model(path, cells: dict, meta: dict) -> None:
    blob = {"version": MODEL_VERSION,
            "created_utc": datetime.now(timezone.utc).strftime("%Y-%m-%dT%H:%M:%SZ"),
            "metric": config.METRIC_COLUMN, "metric_units": config.METRIC_UNITS,
            "policy": policy_snapshot(), **meta,
            "cells": {k: v.to_dict() for k, v in cells.items()}}
    with open(path, "w", encoding="utf-8") as fh:
        json.dump(_clean(blob), fh, indent=2, allow_nan=False)


def read_model(path):
    with open(path, encoding="utf-8") as fh:
        blob = json.load(fh)
    if int(blob.get("version", 0)) != MODEL_VERSION:
        raise ValueError(f"Model file version {blob.get('version')} != "
                         f"{MODEL_VERSION}; refit rather than guess.")
    cells = {k: CellModel.from_dict(v) for k, v in blob.pop("cells").items()}
    return cells, blob
```

```python
"""Charts that let a reviewer see the method rather than take it on trust."""

from __future__ import annotations

import matplotlib

matplotlib.use("Agg")
import matplotlib.pyplot as plt  # noqa: E402
import numpy as np  # noqa: E402
from scipy import stats  # noqa: E402

from cept.features import feature_names  # noqa: E402


def qq_plot(z, path, title: str) -> None:
    z = np.sort(np.asarray(z, float)[np.isfinite(z)])
    q = stats.norm.ppf((np.arange(len(z)) + 0.5) / len(z))
    fig, ax = plt.subplots(figsize=(5, 5))
    ax.plot(q, z, ".", ms=2)
    lim = max(abs(q).max(), 4.5)
    ax.plot([-lim, lim], [-lim, lim], lw=1, color="grey")
    ax.axhline(4, ls="--", lw=0.8, color="red")
    ax.axhline(-4, ls="--", lw=0.8, color="red")
    ax.set(xlabel="normal quantile", ylabel="conditional z", title=title)
    fig.tight_layout()
    fig.savefig(path, dpi=120)
    plt.close(fig)


def coef_plot(cm, path) -> None:
    names = feature_names(cm.include_surprise)
    fig, axes = plt.subplots(1, 2, figsize=(10, 4), sharey=True)
    for ax, coef, ttl in ((axes[0], cm.ls.loc_coef, "expected performance"),
                          (axes[1], cm.ls.sc_coef, "log expected dispersion")):
        ax.barh(names, coef)
        ax.axvline(0, color="grey", lw=0.8)
        ax.set_title(f"{cm.benchmark}: {ttl}\n(effect of +1 sd of condition)")
    fig.tight_layout()
    fig.savefig(path, dpi=120)
    plt.close(fig)
```

```python
"""Readable summaries. The fit summary is the model card's core."""

from __future__ import annotations

import pandas as pd

from cept.features import feature_names
from cept.fit import predicted_count


def _md(df: pd.DataFrame) -> str:
    return df.to_markdown(index=False) if len(df) else "_none_"


def _cells_table(cells: dict) -> pd.DataFrame:
    return pd.DataFrame([{
        "benchmark": b, "n": cm.n,
        "tail_df_lo": round(cm.tail_lo.df, 1), "tail_df_hi": round(cm.tail_hi.df, 1),
        "Tier A lo/hi": f"{cm.thr['A']['lo']:.2f} / {cm.thr['A']['hi']:.2f}",
        "Tier B lo/hi": f"{cm.thr['B']['lo']:.2f} / {cm.thr['B']['hi']:.2f}",
        "expected A per 1,000 orders": round(predicted_count(cm, 1000, "A"), 3),
    } for b, cm in cells.items()])


def _coef_table(cm) -> pd.DataFrame:
    return pd.DataFrame({"condition": feature_names(cm.include_surprise),
                         "centre": cm.ls.loc_coef.round(4),
                         "log dispersion": cm.ls.sc_coef.round(4)})


def write_fit_summary(path, cells, meta, lomo_df, power_df, sens_df,
                      clean_rep, dq_counts) -> None:
    pol = meta.get("policy", {})
    lines = ["# CEPT fit summary (model card)", "",
             f"- market: {meta.get('market')}, fit window "
             f"{meta.get('fit_start')} .. {meta.get('fit_end')}, "
             f"orders {meta.get('n_orders'):,}",
             f"- policy: Tier A rarity {pol.get('RARITY_A')}, floor "
             f"{pol.get('FLOOR_A')}; Tier B rarity {pol.get('RARITY_B')}, "
             f"floor {pol.get('FLOOR_B')}; cohort |t| > {pol.get('COHORT_T')}",
             f"- rationale: {pol.get('POLICY_RATIONALE', '')}", "",
             "## Cells and thresholds (in conditional sigma)", "",
             _md(_cells_table(cells)), "",
             "A tail df of 15 or more means conditional residuals are "
             "near-normal and Tier A is exactly 4 sigma.", ""]
    for b, cm in cells.items():
        lines += [f"## {b}: model coefficients", "", _md(_coef_table(cm)), ""]
    lines += ["## Backtest, leave-one-month-out (predicted vs realised)", "",
              _md(lomo_df), "",
              "## Injected detection (share of planted misses caught)", "",
              _md(power_df), "",
              "## Same-day volatility sensitivity", "",
              f"Orders that would be Tier A without the day-surprise term: "
              f"{len(sens_df)}", "", _md(sens_df), "",
              "## Data quality and cleaning", "",
              _md(pd.DataFrame(list(dq_counts.items()),
                               columns=["rule", "rows"])), "",
              "```", clean_rep.summary(), "```", ""]
    with open(path, "w", encoding="utf-8") as fh:
        fh.write("\n".join(lines))

                        dq_counts) -> None:

    b = scored[scored["tier"] == "B"]
    cols = [c for c in ["order_id", "order_date", "benchmark", "symbol",
                        "perf_in_spreads", "expected_perf", "expected_disp",
                        "z", "excess", "tags"] if c in scored]
    alerts = cohort_df[cohort_df["alert"]] if len(cohort_df) else cohort_df
    lines = ["# CEPT review summary", "",
             f"- model fitted {meta.get('fit_start')} .. {meta.get('fit_end')}",
             f"- orders scored: {len(scored):,}",
             f"- **Tier A (must review): {len(a)}**; Tier B (watch list): {len(b)}",
             f"- cohort alerts: {len(alerts)}",
             f"- data exceptions: {sum(dq_counts.values())}", "",
             "## Tier A", "", _md(a[cols]), "",
             "## Cohort alerts", "", _md(alerts), "",
             "## Backtest this month", "", _md(bt_df), "",
             "## Drift vs fit window", "", _md(drift_df), "",
             "## Data exceptions by rule", "",
             _md(pd.DataFrame(list(dq_counts.items()), columns=["rule", "rows"])),
             ""]
    with open(path, "w", encoding="utf-8") as fh:
        fh.write("\n".join(lines))
```

```python
"""check / fit / score. Fit and score are separate commands with a file
between them: that separation is the leakage guard."""

from __future__ import annotations

import argparse
import os
import sys

import pandas as pd

from cept import (backtest, cohort, config, dq, fit, load, persist, plots,
                  power, report, schema as s, score, sensitivity)
from cept.features import feature_names


def _dq_counts(exc: pd.DataFrame) -> dict:
    return exc["dq_reason"].value_counts().to_dict() if len(exc) else {}


def cmd_check(a) -> int:
    raw = load.read_any(a.csv)
    print(f"file: {a.csv}   rows: {len(raw):,}")
    print("\n--- STRATEGY -> BENCHMARK ---")
    for k, n in raw["Strategy"].astype(str).str.strip().value_counts().items():
        print(f"  {k:<20}{n:>9,}  -> {config.benchmark_for(k) or '*** UNMAPPED ***'}")
    print("\n--- MARKETS ---")
    print(raw["Sym"].astype(str).str[-2:].str.upper().value_counts().to_string())
    df, rep = load.prepare(raw, market=a.market)
    print("\n--- LOADING ---\n" + rep.summary())
    clean, exc = dq.split(df)
    print("\n--- DATA QUALITY ---")
    print(pd.Series(_dq_counts(exc), dtype=int).to_string() if len(exc)
          else "  no data exceptions")
    print("\n--- CONDITIONS (clean rows) ---")
    cols = [c for c in s.CONDITIONS if c != s.SIDE] + [s.FILL_RATE]
    print(clean[[c for c in cols if c in clean]].describe(
        percentiles=[.01, .5, .99]).T.to_string())
    return 0


def cmd_fit(a) -> int:
    df, rep = load.load(a.csv, market=a.market)
    print(rep.summary())
    clean, exc = dq.split(df)
    cells, skipped = fit.fit_all(clean, min_cell_n=a.min_cell_n)
    if not cells:
        print(f"No cell has >= {a.min_cell_n} clean orders: {skipped}",
              file=sys.stderr)
        return 2
    os.makedirs(a.out, exist_ok=True)
    meta = {"market": a.market,
            "fit_start": str(clean[s.ORDER_DATE].min().date()),
            "fit_end": str(clean[s.ORDER_DATE].max().date()),
            "n_orders": int(len(clean)),
            "source_file": os.path.abspath(a.csv),
            "source_hash": persist.file_hash(a.csv), "skipped": skipped,
            "policy": persist.policy_snapshot()}
    persist.write_model(os.path.join(a.out, "model.json"), cells, meta)

    lomo_df = (pd.DataFrame() if a.no_lomo
               else backtest.lomo(clean, min_cell_n=a.min_cell_n))
    power_df = power.injected_detection(cells, clean)
    sens_df = sensitivity.vol_sensitivity(clean, clean, cells)
    scored = score.score_orders(cells, clean)
    coef = pd.concat([pd.DataFrame({"benchmark": b,
                                    "condition": feature_names(cm.include_surprise),
                                    "centre": cm.ls.loc_coef,
                                    "log_dispersion": cm.ls.sc_coef})
                      for b, cm in cells.items()])
    for name, frame in [("lomo_backtest.csv", lomo_df),
                        ("detection_power.csv", power_df),
                        ("vol_sensitivity.csv", sens_df),
                        ("coefficients.csv", coef),
                        ("data_exceptions.csv", exc),
                        ("cleaning_report.csv", rep.to_frame()),
                        ("in_sample_scored.csv", scored)]:
        frame.to_csv(os.path.join(a.out, name), index=False)
    for b, cm in cells.items():
        plots.qq_plot(scored.loc[scored[s.BENCHMARK] == b, "z"],
                      os.path.join(a.out, f"qq_{b}.png"), f"{b} conditional z")
        plots.coef_plot(cm, os.path.join(a.out, f"coef_{b}.png"))
    report.write_fit_summary(os.path.join(a.out, "summary.md"), cells, meta,
                             lomo_df, power_df, sens_df, rep, _dq_counts(exc))
    print(f"\nWrote {os.path.join(a.out, 'model.json')}")
    return 0


def cmd_score(a) -> int:
    cells, meta = persist.read_model(a.model)
    df, rep = load.load(a.csv, market=meta.get("market"))
    print(rep.summary())
    score.check_no_overlap(meta.get("fit_start"), meta.get("fit_end"), df)
    clean, exc = dq.split(df)
    scored = score.score_orders(cells, clean)
    coh = cohort.cohort_test(cells, scored)
    bt = backtest.month_table(cells, scored)
    dr = score.drift(cells, clean)
    os.makedirs(a.out, exist_ok=True)
    a_df = scored[scored["tier"] == "A"].sort_values("excess", ascending=False)
    for name, frame in [("alerts_tierA.csv", a_df),
                        ("watchlist_tierB.csv", scored[scored["tier"] == "B"]),
                        ("scored_orders.csv", scored), ("cohort.csv", coh),
                        ("backtest.csv", bt), ("drift.csv", dr),
                        ("data_exceptions.csv", exc)]:
        frame.to_csv(os.path.join(a.out, name), index=False)
    report.write_score_summary(os.path.join(a.out, "summary.md"), meta, scored,
                               coh, bt, dr, _dq_counts(exc))
    n_coh = int(coh["alert"].sum()) if len(coh) else 0
    print(f"\n  TIER A: {len(a_df)}   TIER B: {int((scored['tier'] == 'B').sum())}"
          f"   COHORT ALERTS: {n_coh}   DATA EXCEPTIONS: {len(exc)}")
    return 0


def build_parser() -> argparse.ArgumentParser:
    p = argparse.ArgumentParser(prog="cept")
    sub = p.add_subparsers(dest="command", required=True)
    c = sub.add_parser("check")
    c.add_argument("--csv", required=True)
    c.add_argument("--market", default=config.DEFAULT_MARKET)
    c.set_defaults(func=cmd_check)
    f = sub.add_parser("fit")
    f.add_argument("--csv", required=True)
    f.add_argument("--out", required=True)
    f.add_argument("--market", default=config.DEFAULT_MARKET)
    f.add_argument("--min-cell-n", type=int, default=config.MIN_CELL_N,
                   dest="min_cell_n")
    f.add_argument("--no-lomo", action="store_true", dest="no_lomo")
    f.set_defaults(func=cmd_fit)
    sc = sub.add_parser("score")
    sc.add_argument("--csv", required=True)
    sc.add_argument("--model", required=True)
    sc.add_argument("--out", required=True)
    sc.set_defaults(func=cmd_score)
    return p


def main(argv: list[str] | None = None) -> int:
    args = build_parser().parse_args(argv)
```
